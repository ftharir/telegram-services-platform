import asyncio
import sys
import uvicorn

from app.core.logging import get_logger
from app.core.shutdown import graceful_shutdown, register_shutdown_callback
from app.database.session import engine
from app.redis_client.client import close_redis
from app.main import main as start_bot
from app.web import app as fastapi_app

logger = get_logger(__name__)


async def close_db_engine() -> None:
    """بستن موتور اتصالات دیتابیس PostgreSQL."""
    logger.info("Disposing database engine connections...")
    await engine.dispose()


async def start_web_server() -> None:
    """راه‌اندازی سرور وب Uvicorn به‌صورت Async."""
    config = uvicorn.Config(
        app=fastapi_app,
        host="0.0.0.0",
        port=8000,
        log_level="info",
    )
    server = uvicorn.Server(config)
    logger.info("Starting Uvicorn Web Server on http://0.0.0.0:8000...")
    await server.serve()


async def main() -> None:
    """اجرای همزمان و موازی ربات تلگرام و وب‌سرور FastAPI همراه با Graceful Shutdown."""
    logger.info("Starting Telegram Services Platform (Bot + Web API)...")
    
    # ثبت توابع پاکسازی منابع دیتابیس و ردیس
    register_shutdown_callback(close_redis)
    register_shutdown_callback(close_db_engine)

    try:
        await asyncio.gather(
            start_bot(),
            start_web_server(),
        )
    except (KeyboardInterrupt, SystemExit):
        logger.info("Termination requested by user/system.")
    except Exception as e:
        logger.critical("Unexpected error during joint execution: %s", e)
    finally:
        await graceful_shutdown("FINAL_CLEANUP")


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except (KeyboardInterrupt, SystemExit):
        pass

import logging
from contextlib import asynccontextmanager
from typing import AsyncGenerator

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from app.api.v1 import api_v1_router
from app.redis_client.client import close_redis, init_redis

logger = logging.getLogger(__name__)


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncGenerator[None, None]:
    """مدیریت رویدادهای شروع (Startup) و پایان (Shutdown) برنامه‌ی FastAPI."""
    logger.info("Starting up FastAPI application...")
    await init_redis()
    yield
    logger.info("Shutting down FastAPI application...")
    await close_redis()


def create_app() -> FastAPI:
    """تابع کارخانه‌ای برای ساخت و پیکربندی اپلیکیشن FastAPI."""
    app = FastAPI(
        title="Telegram Services Platform API",
        description="API بک‌اند برای مدیریت خدمات ربات تلگرام و بررسی سلامت سیستم",
        version="1.0.0",
        lifespan=lifespan,
    )

    # تنظیمات CORS
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    # ثبت روترهای API
    app.include_router(api_v1_router, prefix="/api")

    @app.get("/", tags=["Root"])
    async def root():
        return {
            "message": "Telegram Services Platform API is running!",
            "docs": "/docs",
            "health": "/api/v1/health",
        }

    return app


app = create_app()

import asyncio
import logging
import signal
import sys
from typing import Callable, List

from app.core.logging import get_logger

logger = get_logger(__name__)

# لیستی از توابع پاکسازی که هنگام خاموش شدن باید اجرا شوند
_cleanup_callbacks: List[Callable] = []


def register_shutdown_callback(callback: Callable) -> None:
    """ثبت توابع پاکسازی (مانند بستن اتصالات دیتابیس یا ردیس)."""
    _cleanup_callbacks.append(callback)


async def graceful_shutdown(signal_name: str = "UNKNOWN") -> None:
    """اجرای ترتیبی تمام توابع پاکسازی ثبت‌شده هنگام دریافت سیگنال توقف."""
    logger.info("Received shutdown signal (%s). Starting graceful shutdown...", signal_name)
    
    for callback in _cleanup_callbacks:
        try:
            if asyncio.iscoroutinefunction(callback):
                await callback()
            else:
                callback()
        except Exception as e:
            logger.error("Error executing cleanup callback %s: %s", callback.__name__, e)

    logger.info("Graceful shutdown completed successfully. Exiting process.")


def setup_signal_handlers(loop: asyncio.AbstractEventLoop) -> None:
    """تنظیم شنونده‌های سیگنال‌های SIGINT و SIGTERM در Event Loop."""
    if sys.platform != "win32":
        for sig in (signal.SIGINT, signal.SIGTERM):
            try:
                loop.add_signal_handler(
                    sig,
                    lambda s=sig: asyncio.create_task(graceful_shutdown(s.name))
                )
            except NotImplementedError:
                pass
    else:
        # در ویندوز اضافه کردن signal handler به event loop محدودیت دارد
        logger.debug("Windows environment detected. Signal handling managed via standard asyncio try-finally blocks.")

import logging  
import sys  
from app.core.config import settings  


def setup_logging() -> None:
    """پیکربندی و راه‌اندازی سیستم لاگینگ پروژه."""
    
    log_level = logging.DEBUG if settings.DEBUG else logging.INFO

    
    log_format = "%(asctime)s - [%(levelname)s] - %(name)s - %(message)s"
     
    
    date_format = "%Y-%m-%d %H:%M:%S"

    
    logging.basicConfig(
        level=log_level,  
        format=log_format,  
        datefmt=date_format,  
        handlers=[
            
            logging.StreamHandler(sys.stdout)
        ]
    )
 
    
    logging.getLogger("aiogram").setLevel(logging.INFO)  
    logging.getLogger("asyncio").setLevel(logging.WARNING)  
    logging.getLogger("sqlalchemy.engine").setLevel(logging.WARNING)  

    
    logger = logging.getLogger(__name__)
    
    
    logger.info(f"Logging configured successfully. Environment: {settings.ENVIRONMENT}, Debug: {settings.DEBUG}")


def get_logger(name: str) -> logging.Logger:
    """دریافت یک Logger اختصاصی برای هر ماژول یا فایل.
    
    Args:
        name (str): معمولاً __name__ فایل ارسال می‌شود.
        
    Returns:
        logging.Logger: شیء لاگر آماده برای ثبت پیام.
    """
    
    return logging.getLogger(name)

import logging
from typing import Any, Awaitable, Callable, Dict  

from aiogram import BaseMiddleware  
from aiogram.types import TelegramObject, User as TelegramUser  

from app.database.session import async_session_maker  
from app.services.user_service import UserService  

logger = logging.getLogger(__name__)


class DbSessionMiddleware(BaseMiddleware):
    """میدل‌ور اختصاصی برای مدیریت Session دیتابیس و ثبت‌نام خودکار کاربران."""

    async def __call__(
        self,
        handler: Callable[[TelegramObject, Dict[str, Any]], Awaitable[Any]],
        event: TelegramObject,
        data: Dict[str, Any],
    ) -> Any:
        """این متد قبل از هر درخواست تلگرام اجرا می‌شود."""
        
        
        event_user: TelegramUser | None = data.get("event_from_user")

        
        async with async_session_maker() as session:
            
            user_service = UserService(session)
            data["session"] = session
            data["user_service"] = user_service

            
            if event_user and not event_user.is_bot:
                try:
                    db_user, _ = await user_service.register_or_update_user(
                        telegram_id=event_user.id,
                        first_name=event_user.first_name,
                        username=event_user.username,
                        last_name=event_user.last_name,
                        language_code=event_user.language_code or "fa",
                    )
                    
                    data["db_user"] = db_user
                except Exception as e:
                    logger.error("Error auto-registering user %s: %s", event_user.id, e)

            
            return await handler(event, data)

import logging
from aiogram import Router, F, Bot
from aiogram.filters import CommandStart
from aiogram.types import Message, CallbackQuery

from app.core.config import settings
from app.core.locales import get_text
from app.database.models.user import User
from app.services.user_service import UserService
from app.bot.keyboards import (
    get_main_menu_keyboard,
    get_settings_inline_keyboard,
    get_support_inline_keyboard,
)
from app.bot.middlewares.channel_checker import is_user_member

logger = logging.getLogger(__name__)

router = Router(name="main_router")


@router.message(CommandStart())
async def command_start_handler(message: Message, db_user: User) -> None:
    """هندلر دستور /start - خوش‌آمدگویی و ارائه منوی اصلی."""
    lang = db_user.language_code or "fa"
    welcome_text = get_text("welcome", lang, name=db_user.first_name)
    await message.answer(
        text=welcome_text,
        reply_markup=get_main_menu_keyboard(lang),
    )


@router.callback_query(F.data == "check_channel_join")
async def check_channel_join_callback(callback: CallbackQuery, db_user: User, bot: Bot) -> None:
    """بررسی عضویت هنگام کلیک کاربر روی دکمه «عضو شدم»."""
    lang = db_user.language_code or "fa"
    joined = await is_user_member(bot, callback.from_user.id)

    if joined:
        verified_text = get_text("join_verified", lang)
        if not verified_text or verified_text == "join_verified":
            verified_text = "✅ عضویت شما تایید شد! خوش آمدید."

        await callback.answer(text=verified_text, show_alert=True)
        if callback.message:
            await callback.message.delete()
            await callback.message.answer(
                text=get_text("welcome", lang, name=db_user.first_name),
                reply_markup=get_main_menu_keyboard(lang),
            )
    else:
        not_joined_text = get_text("not_joined_yet", lang)
        if not not_joined_text or not_joined_text == "not_joined_yet":
            not_joined_text = "❌ شما هنوز در کانال عضو نشده‌اید! لطفاً ابتدا عضو شوید."

        await callback.answer(text=not_joined_text, show_alert=True)

@router.callback_query(F.data.startswith("lang_"))
async def change_language_callback(
    callback: CallbackQuery,
    user_service: UserService,
    db_user: User,
) -> None:
    new_lang = callback.data.split("_")[1]
    current_lang = db_user.language_code or "fa"
    
    if current_lang == new_lang:
        await callback.answer(text=get_text("lang_already_active", current_lang), show_alert=True)
        return

    await user_service.change_user_language(db_user.telegram_id, new_lang)
    await callback.answer(text="زبان تغییر یافت / Language updated", show_alert=False)

    if callback.message:
        await callback.message.edit_text(
            text=get_text("lang_changed", new_lang),
            reply_markup=get_settings_inline_keyboard(new_lang),
        )
        await callback.message.answer(
            text=get_text("welcome", new_lang, name=db_user.first_name),
            reply_markup=get_main_menu_keyboard(new_lang),
        )


@router.callback_query(F.data == "close_menu")
async def close_menu_callback(callback: CallbackQuery) -> None:
    if callback.message:
        await callback.message.delete()
    await callback.answer()
